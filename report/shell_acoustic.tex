\paragraph{An acoustic scalar representation.}
Use the entire functions
\[
\mathsf T_j(a,d)=\sum_{m\ge0}\frac{(-a)^md^{2m+j}}{(2m+j)!}.
\]
The momentum constraint gives the scalar amplitude directly, without the hyperbolic
scalar Green function of the unconstrained Cartesian evolution.

\begin{proposition}[Acoustic form of the shell scalar amplitude]\label{prop:shell-acoustic}
On the shell, $c_M\parallel K$ and the longitudinal source is the polynomial
\[
 \mathfrak m\equiv\frac{3iK\cdot c_M}{K^2}
 =\mathsf b'\Big[(1-3\mu)\big(\tfrac32+\mathsf a\big)+\tfrac{3\mu^2-1}4\mathsf b\Big]-6\mathsf a\mathsf a'.
\]
With $a_K=K^2/3=2k^2(1+\mu)/3$, $c_r=k^2(4-\mu)/9$ and $C_N$ from \eqref{eq:moment-CN},
\begin{equation}\label{eq:shell-acoustic}
\boxed{\begin{aligned}
 F_S(t,\mu)={}&18c_r\Big[\mathsf T_2(a_K,t)-\frac{2\mathsf T_3(a_K,t)}t\Big]
 -\int_0^t\mathcal G(t,r)C_N(r,\mu)\dd r\\
 &+(1-3\mu)\big[\tfrac32\mathsf b+3\mathsf a^2-6\mathcal I\big]+\tfrac{3\mu^2-1}8\mathsf b^2-3\mathsf a^2,
\end{aligned}}
\end{equation}
\[
 \mathcal G(t,r)=2\mathsf T_1(a_K,d)+\frac{2[(d-r)\mathsf T_2(a_K,d)-2\mathsf T_3(a_K,d)]}{tr},\qquad
 \mathcal I=\tfrac12\Cin(2x)+\frac{\sin2x}{2x}-\frac{\sin^2x}{2x^2}-\frac12,
\]
with $d=t-r$, $x=k_0t/\sqrt3$ and $\mathcal I=\int_0^t\mathsf a^2/s\,\dd s$. In addition
$F_S'=6(n_2'-n_2/t)+\mathfrak m$ is local in $n_2$.
\end{proposition}
\begin{proof}
Write $v_K=9u_1(Kt/\sqrt3)/K^2$ and let $G_N^+$ be the retarded inverse of $L_N$,
with sign opposite to the source kernel in the triangular scheme. Insert
$n_2=c_rv_K-\int G_N^+C_N$ into $h_S$ of Prop.~\ref{prop:constraint-response}. The inner
integral is elementary,
\[
 \int_r^tG_N^+(s,r)s^{-1}\dd s=\frac{[\mathsf T_3+r\mathsf T_2](a_K,t-r)}{3tr},\qquad
 6v_K-6\int_0^t\frac{v_K}s\dd s=18\Big[\mathsf T_2-\frac{2\mathsf T_3}t\Big](a_K,t).
\] Then $\int_0^t\mathfrak m$ is integrated using
$\mathsf b'=6(\mathsf a'-\mathsf a/t)$.
\end{proof}
Every term in \eqref{eq:shell-acoustic} is entire in $\mu$; its Green frequencies are acoustic,
not hyperbolic. The scalar moments $S_r$ need one time integration and single-variable entire
angular kernels of the type \eqref{eq:moment-N}, plus elementary moments of the polynomial part.
The alternative Cartesian construction, with its nested hyperbolic kernel before constraint
cancellation, is retained in App.~\ref{app:scalar-moments}.
