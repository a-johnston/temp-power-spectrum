\subsubsection{Nine-moment reduction of the contracted cubic fields}\label{sec:cubic-moments}
These field-level shell reductions are used only in the monochromatic spectrum of
\S\ref{sec:mono}, after the complete observable of \S\ref{sec:obs}; nothing in that general
construction relies on them. For the $13$ contraction, the intermediate field equations admit a stronger reduction than
constructing the component kernel $A_3$ and subsequently averaging it. Put
\[
 k=k_0,\quad c=k/\sqrt3,\quad x=ct,\qquad
 \mathsf a=\cos x-\frac{\sin x}{x},\qquad
 \mathsf b=-6\left(\frac{2\sin x}{x}-\cos x-1\right).
\]
Here $\mathsf a,\mathsf b$ are dimensionless first-order profiles, not the background scale
factor. For a shell mode $p$, $\nu_p=\mathsf a$ and
$H_p=\mathsf b(\hat p\hat p-\delta/3)$, and exactly
\begin{equation}\label{eq:cubic-profile-identities}
 L_k\mathsf a=0,\qquad L_k=3\partial_t^2+k^2-6/t^2,\qquad
 \mathsf b'=6(\mathsf a'-\mathsf a/t).
\end{equation}
Write $n_2(t,\mu)=\nu_2(k,q)$, with $F_S,F_T$ the amplitudes of \eqref{eq:shell-two}, and
$n_h(t)=\nu_2(q,-q)$ for the exact homogeneous lapse. Define
\[
 \langle f\rangle_\mu=\tfrac12\int_{-1}^1f(\mu)\dd\mu,\qquad
 q_S=(1+3\mu)/6,\quad q_T=(1-\mu)/2,
\]
\[
 N_r=\langle\mu^r n_2\rangle_\mu,\qquad
 S_r=\langle\mu^r F_S\rangle_\mu,\qquad
 T_r=\langle\mu^r F_T\rangle_\mu,\qquad r=0,1,2.
\]
The moment subscripts here are angular powers, not perturbative orders.

\begin{proposition}[Reduced cubic sources]\label{prop:cubic-moments}
On the shell, the longitudinal cubic momentum source has no contribution made solely of
three first-order fields, pointwise in $\mu$ and at arbitrary finite time. The homogeneous
second-order tensor cancels from both angle-averaged cubic scalar sources; $n_h$ does not.
Those sources require only the nine moments above, their first time derivatives and $n_h$:
\begin{align}
 \bar c_N={}&B_0+\frac{12\mathsf a}{t^2}n_h
 +2\left\langle D_0n_2+\mathsf a k^2\left(\frac{3+\mu}{6}F_S-q_TF_T\right)
 +\frac{\mathsf b'}2(q_SF_S'+q_TF_T')\right\rangle_\mu,\label{eq:cubic-source-N}\end{align}
\begin{equation}\label{eq:cubic-source-M}
 \bar m\equiv\overline{ik\cdot c_M}
 =-2\mathsf a k^2 n_h'
 +2k^2\left\langle\alpha_SF_S'+\alpha_TF_T'+2\mu\mathsf a n_2'
 +\beta_SF_S+\beta_TF_T+\beta_n n_2\right\rangle_\mu,
\end{equation}
where
\[
 B_0=\mathsf a k^2(16+8\mathsf a+2\mathsf b^2/9)-24\mathsf a^3/t^2,\qquad
 D_0=12\mathsf a/t^2-k^2(1+\mu)[5+4\mathsf a+\mathsf b(\mu-1/3)],
\]
\begin{align*}
 \alpha_S&=(3/2+\mathsf a)(3+\mu)/6-\mu\mathsf b q_S/4,&
 \alpha_T&=-(3/2+\mathsf a)q_T-\mu\mathsf b q_T/4,\\
 \beta_S&=(1+\mu)\mathsf b'q_S/4,&
 \beta_T&=(1+\mu)\mathsf b'q_T/4,\\
 \beta_n&=-(1+\mu)[\mathsf b'(\mu-1/3)+2\mathsf a'].
\end{align*}
The factor two counts the pairs $(k,q)$ and $(k,-q)$ after $\mu\mapsto-\mu$; it is not an
additional Wick or factorial normalisation.
\end{proposition}
\begin{proof}
For $M=e^H$, let $C_i=M^{-1}\partial_iM$. Since $\det M=1$ and $\operatorname{tr}S=0$,
\[
 D_jS^j{}_i=\partial_jS^j{}_i+3\zeta_{,j}S^j{}_i-\tfrac12\operatorname{tr}(C_iS).
\]
On the shell $H_1=\mathsf b(t)B(X)$, so $[H_1,H_1']=0$. The quadratic mixed-shear
commutator and the purely first-order cubic double commutator vanish. The remaining cubic
first--second-order commutator is antisymmetric, so its contraction with $k_ik_j$ vanishes.
The quadratic commutator in $C_i$ has zero trace against the symmetric first-order shear.
This removes every direct $111$ term in the longitudinal momentum source, without assuming
$[H_1,H_2]=0$. For the homogeneous tensor, the scalar contractions are proportional to
$\mu^2-1/3$, whose mean is zero. The remaining contractions give
\eqref{eq:cubic-source-N}--\eqref{eq:cubic-source-M}. All weights have degree at most two.
The check \path{solution/check_shell_cubic_sources.py --compare-engine} rebuilds these identities
for arbitrary time profiles and arbitrary pair amplitudes, and compares them with the original
Christoffel-based field engine at arbitrary finite time. It uses no time expansion or quadrature.
\end{proof}

\paragraph{Derivative-free lapse normal form.}
Set
\begin{equation}\label{eq:cubic-normal-form}
 \Upsilon=\mathsf a\langle q_SF_S+q_TF_T\rangle_\mu,\qquad
 \eta_3=\bar\nu_3+\Upsilon,\qquad L_k\eta_3=-\bar r_N.
\end{equation}
Then
\begin{equation}\label{eq:cubic-reduced-source}
\begin{split}
 \bar r_N={}&B_R+\frac{12\mathsf a}{t^2}n_h\\
 &+2\left\langle D_R n_2+\frac{\mathsf a k^2}{6}(2-3\mu-3\mu^2)F_S
 +\frac{\mathsf a k^2}{2}(2+\mu-3\mu^2)F_T\right\rangle_\mu,\\
 D_R={}&12\mathsf a/t^2-k^2(1+\mu)[5+2\mathsf a-6\mathsf a\mu+\mathsf b(\mu-1/3)],\\
 B_R={}&k^2(8\mathsf a-8\mathsf a^2-\tfrac83\mathsf a^2\mathsf b+\tfrac49\mathsf a\mathsf b^2)
 +4\mathsf a\mathsf a'\mathsf b'-24\mathsf a^3/t^2.
\end{split}
\end{equation}
There are no derivatives of any second-order response in this source.
Since $\Upsilon=O(t^4)$, the constrained regular $t^2$ coefficient is unchanged.
The second-order equations used to obtain this normal form are
\begin{equation}\label{eq:moment-pair-equations}
 L_Kn_2=-C_N,\quad
 F_S''+2F_S'/t-K^2F_S/3=-4K^2n_2+J_S,\quad
 F_T''+2F_T'/t+K^2F_T=J_T,
\end{equation}
with $K^2=2k^2(1+\mu)$ and the complete elementary sources
\begin{align}
 C_N={}&k^2[\tfrac23\mathsf a\mathsf b-4\mathsf a
 +\mu(4\mathsf a^2-\tfrac43\mathsf a\mathsf b+2\mathsf a)-2\mu^2\mathsf a\mathsf b]
 +\tfrac12\mathsf b'^2(\mu^2-1/3)+12\mathsf a^2/t^2,\label{eq:moment-CN}\\
 J_S={}&k^2\big[2(3+\mu)\mathsf a^2+(15\mu^2+13\mu-8)\mathsf a\mathsf b/3
 +16(1+\mu)\mathsf a\nonumber\\
 &\qquad+(5-5\mu-15\mu^2-9\mu^3)\mathsf b^2/36
 +(-10+7\mu+15\mu^2)\mathsf b/6+5+7\mu\big]
 -(1+3\mu)\mathsf a'\mathsf b',\label{eq:moment-JS}\\
 J_T={}&\frac{1-\mu}{36}\big\{k^2[\mathsf b^2(9\mu^2+12\mu+1)
 -6\mathsf b(6\mu\mathsf a+3\mu+2)-72\mathsf a^2+36]-36\mathsf a'\mathsf b'\big\}.
 \label{eq:moment-JT}
\end{align}
In particular
\[
 \langle q_SJ_S+q_TJ_T\rangle_\mu
 =\tfrac2{27}[k^2(12\mathsf a\mathsf b+72\mathsf a-\mathsf b^2+36)-18\mathsf a'\mathsf b'],
\]
which evaluates the elementary forcing part in $B_R$.

The derivative cancellation also holds at general momenta. For a partition $p|(q,r)$,
put $P=q+r$, $K=p+P$, $Q_p=pp-\delta p^2/3$, $J=H_{qr}$ and
$\phi_{p|qr}=\nu_p Q_p:J/(2p^2)$. The metric-dependent source
$-\nu_pK_iJ_{ij}p_j+\tfrac12H_p':J'$ equals $L_K\phi_{p|qr}$ plus
\[
 -\nu_pK_iJ_{ij}p_j-\frac{\nu_p}{2p^2}
 [(K^2-p^2)Q_p:J+3Q_p:(J''+2J'/t)].
\]
Use the second-order spatial equation for the acceleration and sum the three partitions.
There is then no derivative of $J$. At $p=0$ use the regular continuation of
$\nu_p/p^2$, since the regular scalar has $\nu_p=O(p^2)$ at fixed time.
The change of variable must be undone when reconstructing the observable.
